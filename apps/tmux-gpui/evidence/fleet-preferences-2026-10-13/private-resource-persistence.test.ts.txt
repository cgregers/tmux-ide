import { test } from "node:test";
import assert from "node:assert/strict";
import { mkdtemp, readFile, rm, writeFile, readdir } from "node:fs/promises";
import { join } from "node:path";
import { randomUUID } from "node:crypto";
import { Hono } from "hono";
import { serve } from "@hono/node-server";
import { mountFleetClientStateRoute } from "../../packages/daemon/src/command-center/resources/fleet-client-state-route.ts";
import { loadFleetClientState, updateFleetClientState } from "../../packages/daemon/src/lib/fleet-client-state.ts";
import { createFleetPreferences } from "../../apps/tmux-gpui/bridge/fleet-preferences.ts";
import { createLocalFleetPreferenceOwner } from "../../apps/tmux-gpui/bridge/fleet-preferences-owner.ts";

test("real fleet resource persists adapter writes and preserves malformed storage", { timeout: 15000 }, async () => {
  const home = await mkdtemp("/private/tmp/gpui-fleet-persistence-");
  const path = join(home, "fleet-view.json");
  const daemon = { instanceId: randomUUID(), startedAt: new Date().toISOString(), protocolVersion: 2, productVersion: "private-qualification" };
  const token = randomUUID();
  const app = new Hono();
  let posts = 0;
  app.use("*", async (c, next) => { if (c.req.method === "POST") posts++; await next(); });
  mountFleetClientStateRoute(app, { daemon, ownerToken: token, read: () => loadFleetClientState(path), update: change => updateFleetClientState(change, path) });
  let server: ReturnType<typeof serve> | undefined;
  let adapter: ReturnType<typeof createFleetPreferences> | undefined;
  try {
    const port = await new Promise<number>((resolve, reject) => {
      server = serve({ fetch: app.fetch, hostname: "127.0.0.1", port: 0 }, info => resolve(info.port));
      server.on("error", reject);
    });
    const url = `http://127.0.0.1:${port}/api/resources/fleet-client-state`;
    const info = { ...daemon, port, pid: process.pid, bindHostname: "127.0.0.1" as const, authToken: token };
    const handle = {
      id: "local", kind: "local" as const, label: "Private fixture", ready: Promise.resolve(true),
      read: () => info, isAlive: async () => true, observe: async () => () => {},
      endpoint: () => ({ kind: "local" as const, label: null, remote: null, localBaseUrl: null, epoch: 0, state: "ready" as const }),
    };
    const getLocalOwner = createLocalFleetPreferenceOwner({ getMachine(id) { assert.equal(id, "local"); return handle; } });
    adapter = createFleetPreferences({ getLocalOwner });
    const rejected = await fetch(url, { signal: AbortSignal.timeout(3000) });
    assert.notEqual(rejected.status, 200); await rejected.body?.cancel();
    assert.equal(await adapter.refresh(), true);
    assert.deepEqual(adapter.snapshot().state?.favorites, []);
    for (const change of [
      { type: "favorite", key: "session-a", enabled: true },
      { type: "favorite", key: "session-a", enabled: true },
      { type: "favorite", key: "session-b", enabled: true },
      { type: "collapse", key: "environment-one", enabled: true },
    ]) assert.equal(await adapter.change(adapter.snapshot().revision, change), true);
    assert.deepEqual(loadFleetClientState(path).favorites, ["session-a", "session-b"]);
    adapter.dispose(); adapter = createFleetPreferences({ getLocalOwner });
    assert.equal(await adapter.refresh(), true);
    assert.deepEqual(adapter.snapshot().state?.collapsed, ["environment-one"]);
    // Same daemon storage reader/reducer used by the TUI; this is not a GUI test.
    updateFleetClientState({type: "favorite", key: "session-a", enabled: false}, path);
    assert.equal(await adapter.refresh(), true);
    assert.deepEqual(adapter.snapshot().state?.favorites, ["session-b"]);
    const before = await readFile(path, "utf8");
    const stale = await fetch(url, { method: "POST", headers: {Authorization: `Bearer ${token}`, "Content-Type": "application/json"}, body: JSON.stringify({expectedInstanceId: randomUUID(), change: {type: "collapse", key: "environment-one", enabled: false}}), signal: AbortSignal.timeout(3000) });
    assert.equal(stale.status, 409); await stale.body?.cancel();
    assert.equal(await readFile(path, "utf8"), before);
    await writeFile(path, "malformed fixture state\n");
    const count = posts;
    assert.equal(await adapter.change(adapter.snapshot().revision, {type: "favorite", key: "session-c", enabled: true}), false);
    assert.equal(posts, count + 1);
    assert.equal(await adapter.refresh(), false);
    assert.equal(adapter.snapshot().state, null);
    assert.equal(await readFile(path, "utf8"), "malformed fixture state\n");
    assert.deepEqual(await readdir(home), ["fleet-view.json"]);
  } finally {
    adapter?.dispose();
    if (server) {
      server.closeAllConnections();
      await new Promise<void>((resolve, reject) => server!.close(error => error ? reject(error) : resolve()));
    }
    await rm(home, { recursive: true, force: true });
  }
});
