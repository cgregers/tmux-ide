// Private raw-PTY receipt for operator-driven OS dead-key composition.
// Exact bytes alone do not prove the OS composition path; preserve marked-UI evidence separately.
import assert from "node:assert/strict";
import { execFileSync, spawn } from "node:child_process";
import { access, mkdir, writeFile, readFile, stat } from "node:fs/promises";
import { constants, writeFileSync } from "node:fs";
import { join, isAbsolute } from "node:path";
import { createHash, randomUUID } from "node:crypto";
import { Buffer } from "node:buffer";
import process from "node:process";
import console from "node:console";
import { setTimeout as delay } from "node:timers/promises";
import { createScratchFleet } from "../../scripts/lib/product-fixtures/scratch-fleet.ts";
import { stopFixtureChild } from "../../apps/tmux-gpui/bridge/fixture-child.mjs";

const appPath = process.env.TMUX_GPUI_TEST_APP;
const directory = process.env.TMUX_GPUI_COMPOSITION_DIR;
assert.ok(appPath && isAbsolute(appPath), "Require absolute TMUX_GPUI_TEST_APP");
assert.ok(directory && isAbsolute(directory), "Require new absolute TMUX_GPUI_COMPOSITION_DIR");
const launcher = join(appPath, "Contents/MacOS/tmux-ide-launcher");
assert.ok((await stat(launcher)).isFile(), "App launcher must be a regular file");
await access(launcher, constants.X_OK);
const privateCli = process.env.TMUX_GPUI_TEST_CLI;
assert.ok(privateCli, "Require explicit TMUX_GPUI_TEST_CLI");
{
  assert.ok(isAbsolute(privateCli), "TMUX_GPUI_TEST_CLI must be absolute");
  assert.ok((await stat(privateCli)).isFile(), "Private CLI must be a regular file");
  await access(privateCli, constants.R_OK);
}
let fatal;
const cancel = () => {
  fatal ??= new Error("Native composition proof cancelled");
};
process.on("SIGINT", cancel);
process.on("SIGTERM", cancel);
async function startFixtureDaemon(fleet) {
  if (fatal) throw fatal;
  const child = spawn(process.execPath, [privateCli, "--headless"], {
    env: { ...process.env, ...fleet.environment },
    stdio: "ignore",
  });
  let childError;
  child.on("error", (error) => {
    childError = error;
  });
  try {
    const deadline = Date.now() + 45000;
    for (;;) {
      if (fatal) throw fatal;
      if (childError) throw childError;
      assert.ok(child.exitCode === null && child.signalCode === null, "Private daemon exited");
      assert.ok(Date.now() < deadline, "Private daemon startup deadline");
      let record;
      try {
        record = JSON.parse(await readFile(join(fleet.daemonInfoDir, "daemon.json"), "utf8"));
      } catch {
        /* Await the owned atomic discovery record. */
      }
      if (record?.pid === child.pid && typeof record.authToken === "string" && record.instanceId)
        return { record, stop: () => stopFixtureChild(child) };
      await delay(20);
    }
  } catch (error) {
    try {
      await stopFixtureChild(child);
    } catch (cleanupError) {
      throw new AggregateError([error, cleanupError], "Private daemon startup failed", {
        cause: cleanupError,
      });
    }
    throw error;
  }
}
const payload = "\u00e9";
const expected = Buffer.from(payload, "utf8");
const actualPath = join(directory, "actual.bin");
const producerPath = join(directory, "producer.cjs");
const quote = (value) => "'" + value.replaceAll("'", "'\\''") + "'";
const digest = (bytes) => createHash("sha256").update(bytes).digest("hex");
for (const key of Object.keys(process.env)) {
  if (
    key.startsWith("TMUX_IDE_") ||
    ["TMUX", "TMUX_PANE", "TMUX_TMPDIR", "NODE_OPTIONS", "NODE_PATH"].includes(key)
  )
    delete process.env[key];
}
// Exclusive mkdir: never reuse or remove caller evidence, including on failure.
await mkdir(directory, { mode: 0o700 });
let fleet, daemon, app, closedResult, result, producerPid;
const failures = [];
const deadline = Date.now() + 600000;
const producer = `const fs = require('node:fs');
const output = process.argv[2];
fs.writeFileSync(output, Buffer.alloc(0), {flag:'wx',mode:0o600});
if (!process.stdin.isTTY) throw new Error('Private pane requires TTY');
process.stdin.setRawMode(true);
process.stdin.resume();
let received = 0;
process.stdin.on('data', chunk => {
  if (received + chunk.length > 4096) { process.exitCode=1; process.stdin.destroy(); return; }
  fs.appendFileSync(output, chunk);
  received += chunk.length;
  process.stdout.write('\\r\\nRECEIVED_BYTES=' + received + '\\r\\n');
});
process.stdout.write('\\x1b[?2004l\\x1b[2J\\x1b[HCOMPOSITION_READY: use OS dead-key composition once; no paste or Return.\\r\\n');
`;
const run = (...args) =>
  execFileSync(fleet.environment.TMUX_IDE_TMUX_BIN, ["-S", fleet.socketPath, ...args], {
    env: { ...process.env, ...fleet.environment },
    encoding: "utf8",
    timeout: 5000,
  }).trimEnd();
async function wait(predicate, label, allowClosed = false) {
  for (;;) {
    if (fatal) throw fatal;
    if (!allowClosed && app && (app.exitCode !== null || app.signalCode !== null))
      throw new Error("App closed before composition proof");
    const done = await predicate();
    if (fatal) throw fatal;
    if (done) return;
    assert.ok(Date.now() < deadline, `Native composition deadline: ${label}`);
    await delay(40);
  }
}
async function actual(path = actualPath) {
  const info = await stat(path);
  assert.ok(info.isFile() && info.size <= 4096, "Bounded private byte capture");
  return readFile(path);
}
try {
  if (fatal) throw fatal;
  await writeFile(join(directory, "payload.txt"), payload, { flag: "wx", mode: 0o600 });
  await writeFile(join(directory, "expected.bin"), expected, { flag: "wx", mode: 0o600 });
  await writeFile(producerPath, producer, { flag: "wx", mode: 0o600 });
  fleet = await createScratchFleet({
    sessions: 1,
    windowsPerSession: 1,
    slug: "gpui-native-composition",
  });
  if (fatal) throw fatal;
  const pane = fleet.initialPanes[0].paneId;
  const secondPath = join(directory, "actual-b.bin");
  run("respawn-pane", "-k", "-t", pane, `exec ${quote(process.execPath)} ${quote(producerPath)} ${quote(actualPath)}`);
  const paneB = run("split-window", "-h", "-t", pane, "-P", "-F", "#{pane_id}",
    `exec ${quote(process.execPath)} ${quote(producerPath)} ${quote(secondPath)}`);
  assert.match(paneB, /^%[0-9]+$/);
  const pids = [];
  for (const target of [pane, paneB]) {
    await wait(() => run("capture-pane", "-p", "-t", target).includes("COMPOSITION_READY"), "raw producer ready");
    assert.equal(run("display-message", "-p", "-t", target, "#{bracket_paste_flag}"), "0");
    const pid = run("display-message", "-p", "-t", target, "#{pane_pid}");
    assert.match(pid, /^[1-9][0-9]*$/);
    pids.push(pid);
  }
  producerPid = pids[0];
  const empty = Buffer.alloc(0), x = Buffer.from("x");
  async function exact(a, b) {
    assert.deepEqual(await actual(), a, "Pane A exact bytes");
    assert.deepEqual(await actual(secondPath), b, "Pane B exact bytes");
  }
  async function checkpoint(stage, a, b, instruction) {
    const nonce = randomUUID();
    const confirmationPath = join(directory, stage + ".confirm");
    console.log(JSON.stringify({stage, nonce, confirmationPath, paneA: pane, paneB, instruction}));
    await wait(async () => {
      await exact(a, b);
      let value;
      try { value = await readFile(confirmationPath, "utf8"); }
      catch (error) { if (error.code === "ENOENT") return false; throw error; }
      assert.equal(value, nonce, "Exact current operator nonce required");
      return true;
    }, stage);
  }
  await exact(empty, empty);
  daemon = await startFixtureDaemon(fleet);
  if (fatal) throw fatal;
  app = spawn(launcher, [], {
    cwd: fleet.root,
    env: { ...process.env, ...fleet.environment, PATH: "/usr/bin:/bin" },
    stdio: ["ignore", "inherit", "inherit"],
  });
  app.on("error", (error) => {
    fatal ??= error;
  });
  app.once("close", (code, signal) => {
    closedResult = { code, signal };
  });
  await checkpoint("mark-a", empty, empty,
    "Open session, click LEFT pane A, Option+E only. Capture marked UI, then write nonce; no paste/Return.");
  console.log("Now press E to commit é to pane A exactly once.");
  await wait(async () => {
    const a = await actual(), b = await actual(secondPath);
    assert.deepEqual(b, empty);
    assert.ok(a.length <= expected.length && expected.subarray(0,a.length).equals(a));
    return a.equals(expected);
  }, "A single commit");
  await checkpoint("commit-a", expected, empty,
    "Capture committed é evidence; confirm before starting the next marked input.");
  await checkpoint("remark-a", expected, empty,
    "In A press Option+E only again; capture marked UI and confirm zero additional bytes.");
  console.log("Now click RIGHT pane B and type plain x only. Do not commit or paste in A.");
  await wait(async () => {
    assert.deepEqual(await actual(), expected);
    const b = await actual(secondPath);
    assert.ok(b.length <= 1 && x.subarray(0,b.length).equals(b));
    return b.equals(x);
  }, "B exact attribution");
  await checkpoint("attribution-b", expected, x,
    "Capture pane B x and cancelled/uncommitted A mark; confirm exact attribution, then Cmd-Q.");
  console.log("Close only this owned app with Cmd-Q.");
  await wait(async () => { await exact(expected,x); return closedResult !== undefined; }, "operator Cmd-Q", true);
  assert.equal(closedResult.code, 0, "Clean native close");
  for (const [index,target] of [pane,paneB].entries()) {
    assert.equal(run("display-message", "-p", "-t", target, "#{pane_pid}"), pids[index]);
    assert.equal(run("display-message", "-p", "-t", target, "#{pane_dead}"), "0");
  }
  await exact(expected, x);
  await writeFile(join(directory,"source-after-close.txt"),
    [pane,paneB].map(target => run("capture-pane","-p","-t",target)).join("\n---\n"),
    {flag:"wx",mode:0o600});
  result = {
    passed: true,
    rawCompositionBytesVerified: true,
    twoPaneAttributionVerified: true,
    paneAHex: expected.toString("hex"),
    paneBHex: "78",
    osCompositionPathRequiresOperatorEvidence: true,
    bracketPasteFlag: 0,
    exactBytes: expected.length,
    sha256: digest(expected),
    sourceSurvivesClose: true,
    producerPid,
  };
} catch (error) {
  failures.push(error);
} finally {
  // Independent cleanup attempts; the private fleet owns and reaps the raw producer.
  for (const cleanup of [
    () => stopFixtureChild(app),
    () => daemon?.stop(),
    () => fleet?.dispose(),
  ]) {
    try {
      await cleanup();
    } catch (error) {
      failures.push(error);
    }
  }
}
if (fatal) failures.push(fatal);
// Commit the small receipt in the same turn as the final cancellation check.
writeFileSync(
  join(directory, "receipt.json"),
  JSON.stringify(
    failures.length
      ? { passed: false, failureCount: failures.length, expectedSha256: digest(expected) }
      : { ...result, cleanup: true },
    null,
    2,
  ) + "\n",
  { flag: "wx", mode: 0o600 },
);
if (failures.length) throw new AggregateError(failures, "Native composition fixture failed");
console.log(JSON.stringify({ ...result, cleanup: true }));
